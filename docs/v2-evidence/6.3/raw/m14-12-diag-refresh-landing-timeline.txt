/**
 * TEMPORARY task 6.3 M14 diagnostic — deleted after its run; source kept in
 * docs/v2-evidence/6.3/raw/m14-12-*.
 *
 * Measures what a refresh on Personal paints, and when: the step line is sampled
 * every animation frame from the first paint after the reload until it has held
 * one value for 3 s, together with the time the assignment preview answered.
 * Root-causing F3's one failure (m14-11): the hypothesis is that the resume
 * machine paints the derived resume step (Education) first and the address bar's
 * `?step=personal` moves the wizard to Personal only once the preview resolves —
 * so a helper that reads the step in that window and clicks "Step 3: Personal"
 * can click into the move.
 */
import { test, expect } from '../harness-fixtures';
import { createDraftApplication, superAdminContext, type AuthedContext } from '../helpers/api';
import {
  allowHarnessNoise,
  configureWizardOffering,
  loginStudentInPage,
  openPersonalStep,
  waitForNetworkQuiet,
  wizardStepLine,
} from '../helpers/wizard-draft';

let sharedAdmin: Promise<AuthedContext> | null = null;
function platformAdmin(): Promise<AuthedContext> {
  sharedAdmin ??= superAdminContext();
  return sharedAdmin;
}
test.afterAll(async () => {
  if (sharedAdmin) {
    const admin = await sharedAdmin;
    await admin.ctx.dispose();
    sharedAdmin = null;
  }
});

test('diagnostic: what a refresh on Personal paints, and when', async ({ page, gate }) => {
  test.setTimeout(180_000);
  allowHarnessNoise(gate);
  const offering = await configureWizardOffering(await platformAdmin(), 'Diag Landing', 'T63D');
  const { email, csrf } = await loginStudentInPage(page);
  const { applicationId } = await createDraftApplication(page.request, csrf, {
    canonicalProgramId: offering.canonicalProgramId,
    intakeId: offering.intakeId,
    email,
  });
  await page.goto(`/student/application-wizard?mode=resume&draftId=${encodeURIComponent(applicationId)}`, {
    waitUntil: 'domcontentloaded',
  });
  await expect(page.getByTestId('wizard-action-bar')).toBeVisible({ timeout: 60_000 });
  await openPersonalStep(page);
  await waitForNetworkQuiet(page);
  expect(new URL(page.url()).searchParams.get('step')).toBe('personal');

  await page.addInitScript(() => {
    const samples: Array<[number, string]> = [];
    (window as unknown as { __m14StepSamples: typeof samples }).__m14StepSamples = samples;
    const t0 = performance.now();
    const sample = () => {
      const line = document.querySelector('[data-testid="wizard-progress-summary"] p');
      const text = line?.textContent?.trim() ?? '(none)';
      if (samples.length === 0 || samples[samples.length - 1]![1] !== text) {
        samples.push([Math.round(performance.now() - t0), text]);
      }
      requestAnimationFrame(sample);
    };
    requestAnimationFrame(sample);
  });
  const navStart = Date.now();
  const previewAt: number[] = [];
  page.on('response', (response) => {
    if (new URL(response.url()).pathname.startsWith('/api/v1/catalog/assignment-preview/')) {
      previewAt.push(Date.now() - navStart);
    }
  });
  const readAt: number[] = [];
  page.on('response', (response) => {
    if (new URL(response.url()).pathname === `/api/v1/applications/${applicationId}/`) {
      readAt.push(Date.now() - navStart);
    }
  });
  await page.reload({ waitUntil: 'domcontentloaded' });
  await expect(wizardStepLine(page)).toHaveText('Step 3 of 6', { timeout: 60_000 });
  await page.waitForTimeout(3_000);
  const samples = await page.evaluate(
    () => (window as unknown as { __m14StepSamples: Array<[number, string]> }).__m14StepSamples,
  );
  const firstStep = samples.find(([, text]) => /^Step \d of 6$/.test(text));
  const firstPersonal = samples.find(([, text]) => text === 'Step 3 of 6');
  test.info().annotations.push({
    type: 'refresh on Personal: step-line timeline (ms since the document started)',
    description: JSON.stringify({
      samples,
      firstStep,
      firstPersonal,
      transientMs: firstStep && firstPersonal ? firstPersonal[0] - firstStep[0] : null,
      draftRowReadAtMsSinceReload: readAt,
      assignmentPreviewAnsweredAtMsSinceReload: previewAt,
    }),
  });
});
