# Source of the temporary diagnostic spec (removed from the tree after this run)
/**
 * TEMPORARY diagnostic for task 6.3 M13 (deleted after the run; output kept in
 * docs/v2-evidence/6.3/raw/m13-03-*). Not a required journey.
 *
 * Question: M12 two-tab run 3 failed after "Reload latest draft" with the page on
 * /auth/signin. The host suspended for 5h42m at that click, so the 30-minute
 * access cookie expired while the 7-day refresh cookie stayed valid. Does a page
 * load in that cookie state sign the student out? This reproduces the cookie
 * state deterministically by dropping only the access cookie.
 */
import { test, expect, type Page } from '@playwright/test';
import {
  assignOffering,
  createDraftApplication,
  createFee,
  createInstitutionViaApi,
  createIntake,
  linkIntake,
  listCanonicalPrograms,
  registerAndLoginStudent,
  superAdminContext,
} from '../helpers/api';

test.describe.configure({ mode: 'serial' });

function unwrap(json: unknown): Record<string, unknown> {
  const body = (json ?? {}) as { data?: unknown };
  return ((body.data ?? body) as Record<string, unknown>) ?? {};
}

async function setup(page: Page) {
  const admin = await superAdminContext();
  const stamp = `${Date.now()}`.slice(-8);
  const canonical = (await listCanonicalPrograms(admin))[0];
  const { institutionId } = await createInstitutionViaApi(admin, {
    name: `Diag School ${stamp}`,
    code: `DG${stamp}`,
    hostname: `dg-${stamp}.localhost`,
  });
  const { intakeId } = await createIntake(admin, { name: `Diag Intake ${stamp}`, year: 2027 });
  const { offeringId } = await assignOffering(admin, institutionId, {
    canonical_program_id: canonical.id,
    name: canonical.name,
    code: `DGP${stamp}`,
  });
  await linkIntake(admin, { program_id: offeringId, intake_id: intakeId, max_capacity: 25 });
  await createFee(admin, offeringId, { amount: '153.00' });
  await admin.ctx.dispose();

  const { ctx, email, password } = await registerAndLoginStudent();
  await ctx.dispose();
  const login = await page.request.post('/api/v1/auth/login/', { data: { email, password } });
  expect(login.ok()).toBeTruthy();
  const csrf = login.headers()['x-csrf-token'] || '';
  const { applicationId } = await createDraftApplication(page.request, csrf, {
    canonicalProgramId: canonical.id as string,
    intakeId,
    email,
  });
  return { applicationId, csrf };
}

function recordAuthTraffic(page: Page): string[] {
  const seen: string[] = [];
  page.on('response', (response) => {
    const path = new URL(response.url()).pathname;
    if (path.startsWith('/api/v1/auth/')) {
      seen.push(`${response.status()} ${response.request().method()} ${path}${new URL(response.url()).search}`);
    }
  });
  return seen;
}

async function landing(page: Page): Promise<string> {
  const actionBar = page.getByTestId('wizard-action-bar');
  const signIn = page.getByRole('heading', { name: /sign in to your application/i });
  await expect(actionBar.or(signIn)).toBeVisible({ timeout: 60_000 });
  // Let the session bootstrap settle; a guard redirect can follow the first paint.
  await page.waitForTimeout(6_000);
  if (await signIn.isVisible()) return `SIGN-IN PAGE (${new URL(page.url()).pathname})`;
  if (await actionBar.isVisible()) return `WIZARD (${new URL(page.url()).pathname})`;
  return `OTHER (${page.url()})`;
}

test('D1 control: reload with a valid access cookie reopens the wizard', async ({ page }) => {
  test.setTimeout(180_000);
  const { applicationId } = await setup(page);
  await page.goto(`/student/application-wizard?mode=resume&draftId=${applicationId}`);
  await expect(page.getByTestId('wizard-action-bar')).toBeVisible({ timeout: 60_000 });
  const traffic = recordAuthTraffic(page);
  await page.reload({ waitUntil: 'domcontentloaded' });
  const where = await landing(page);
  console.log(`DIAG D1 landing after reload with valid access cookie: ${where}`);
  console.log(`DIAG D1 auth traffic: ${JSON.stringify(traffic)}`);
});

test('D2: reload after only the access cookie expired', async ({ page }) => {
  test.setTimeout(180_000);
  const { applicationId } = await setup(page);
  await page.goto(`/student/application-wizard?mode=resume&draftId=${applicationId}`);
  await expect(page.getByTestId('wizard-action-bar')).toBeVisible({ timeout: 60_000 });
  const before = (await page.context().cookies()).map((c) => c.name).sort();
  await page.context().clearCookies({ name: 'access_token' });
  const after = (await page.context().cookies()).map((c) => c.name).sort();
  console.log(`DIAG D2 cookies before=${JSON.stringify(before)} after=${JSON.stringify(after)}`);
  const traffic = recordAuthTraffic(page);
  await page.reload({ waitUntil: 'domcontentloaded' });
  const where = await landing(page);
  console.log(`DIAG D2 landing after reload with expired access cookie: ${where}`);
  console.log(`DIAG D2 auth traffic during the reload: ${JSON.stringify(traffic)}`);
  const refresh = await page.request.post('/api/v1/auth/refresh/');
  console.log(`DIAG D2 the refresh cookie was still valid: POST /auth/refresh/ -> ${refresh.status()}`);
});

test('D3: conflict, then "Reload latest draft" after the access cookie expired', async ({ page }) => {
  test.setTimeout(180_000);
  const { applicationId, csrf } = await setup(page);
  const draftPath = `/api/v1/applications/drafts/${applicationId}/`;
  await page.goto(`/student/application-wizard?mode=resume&draftId=${applicationId}`);
  await expect(page.getByTestId('wizard-action-bar')).toBeVisible({ timeout: 60_000 });
  // Reach the Personal step (resume derives a later step).
  const summary = page.getByTestId('wizard-progress-summary');
  await expect(summary).toContainText(/Step [234] of 6/, { timeout: 60_000 });
  const personal = page.getByRole('button', { name: /Step 3: Personal/i });
  if (!(await summary.innerText()).includes('Step 3 of 6')) {
    const toPersonal = page.getByRole('button', { name: /continue to personal/i });
    if (await toPersonal.isVisible()) await toPersonal.click();
    else await personal.click();
  }
  const town = page.getByLabel(/town|city/i).first();
  await expect(town).toBeVisible({ timeout: 60_000 });

  // Another device wins: a canonical versioned write the open tab does not know about.
  const current = unwrap(await (await page.request.get(draftPath)).json());
  const won = await page.request.patch(draftPath, {
    headers: { 'x-csrf-token': csrf, 'Idempotency-Key': `diag-${Date.now()}` },
    data: { residence_town: `Diag winner ${Date.now()}`, expected_version: Number(current.version) },
  });
  console.log(`DIAG D3 other-device write -> ${won.status()}`);

  // The open tab edits on the old version: its own autosave meets the conflict.
  const stale = page.waitForResponse(
    (r) => r.request().method() === 'PATCH' && new URL(r.url()).pathname === draftPath,
    { timeout: 30_000 },
  );
  await town.fill(`Diag stale ${Date.now()}`);
  console.log(`DIAG D3 stale autosave -> ${(await stale).status()}`);
  const reloadLatest = page.getByRole('button', { name: /reload latest draft/i });
  await expect(reloadLatest).toBeEnabled();

  // The student leaves it there for more than 30 minutes: the access cookie lapses.
  await page.context().clearCookies({ name: 'access_token' });
  const traffic = recordAuthTraffic(page);
  const navigated = page.waitForNavigation({ waitUntil: 'domcontentloaded' });
  await reloadLatest.click();
  await navigated;
  const where = await landing(page);
  console.log(`DIAG D3 landing after "Reload latest draft" with expired access cookie: ${where}`);
  console.log(`DIAG D3 auth traffic during the reload: ${JSON.stringify(traffic)}`);
});
