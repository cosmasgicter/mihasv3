# reporter, 4 mobile repeats, all passed) put the extra time in ONE step: the 'Reload latest draft' click, 6.2-9.9 s.
# - Single tab (DIAG #1-#3): click() returns in 58-71 ms and the navigation commits 3-5 ms later; the page clock shows
#   click -> outbox removal -> beforeunload within 6 ms. No confirm dialog (dialogs=0, defaultPrevented=false).
# - Background tab, a second page open in front (DIAG-BG #1-#3): Playwright's click() takes 4.0-7.8 s to return, but the
#   page's own clock shows click -> beforeunload in 4-5 ms. The delay is before the click reaches the page: it is how
#   long the driver takes to deliver input to a non-front page, not app work. The app's reload is immediate in both cases.
# Conclusion: instrument characteristic, not a product defect. The two-tab journey's timeouts (60 s) are well above it.

/**
 * TEMPORARY diagnostic for task 6.3 M13 (deleted after the run; output kept in
 * docs/v2-evidence/6.3/raw/m13-11-*). Not a required journey.
 *
 * Question: in the two-tab journey the "Reload latest draft" click takes 6-10 s
 * before the navigation starts, although `adapters.reload` waits at most 1.5 s
 * for the storage flush. Where does the time go? The instrument timestamps (page
 * clock): the click, `beforeunload` (and whether it asked), `location.reload`,
 * `requestIdleCallback` scheduling/firing, and every localStorage write/removal
 * of the draft keys.
 */
import { test, expect, type Page } from '@playwright/test';
import {
  assignOffering,
  createDraftApplication,
  createFee,
  createInstitutionViaApi,
  createIntake,
  linkIntake,
  listCanonicalPrograms,
  registerAndLoginStudent,
  superAdminContext,
} from '../helpers/api';

function unwrap(json: unknown): Record<string, unknown> {
  const body = (json ?? {}) as { data?: unknown };
  return ((body.data ?? body) as Record<string, unknown>) ?? {};
}

async function openPersonalStep(page: Page): Promise<void> {
  const summary = page.getByTestId('wizard-progress-summary');
  const town = page.getByLabel(/town|city/i).first();
  for (let attempt = 0; attempt < 3; attempt += 1) {
    await expect(summary).toContainText(/Step [234] of 6/, { timeout: 60_000 });
    const currentStep = await summary.locator('p').first().textContent();
    if (currentStep?.includes('Step 2 of 6')) {
      await page.getByRole('button', { name: /continue to personal/i }).click();
    } else if (!currentStep?.includes('Step 3 of 6')) {
      await page.getByRole('button', { name: /Step 3: Personal/i }).click();
    }
    try {
      await expect(town).toBeVisible({ timeout: 10_000 });
      await expect(summary).toContainText('Step 3 of 6');
      return;
    } catch {
      // retry
    }
  }
  await expect(town).toBeVisible({ timeout: 60_000 });
}

for (const repeat of [1, 2, 3]) {
  test(`reload-latest latency from a BACKGROUND tab #${repeat}`, async ({ page }) => {
    test.setTimeout(180_000);
    const admin = await superAdminContext();
    const stamp = `${Date.now()}`.slice(-8);
    const canonical = (await listCanonicalPrograms(admin))[0];
    const { institutionId } = await createInstitutionViaApi(admin, {
      name: `Reload BG Diag ${stamp}`,
      code: `RB${stamp}`,
      hostname: `rb-${stamp}.localhost`,
    });
    const { intakeId } = await createIntake(admin, { name: `Reload BG Intake ${stamp}`, year: 2027 });
    const { offeringId } = await assignOffering(admin, institutionId, {
      canonical_program_id: canonical.id,
      name: canonical.name,
      code: `RBP${stamp}`,
    });
    await linkIntake(admin, { program_id: offeringId, intake_id: intakeId, max_capacity: 25 });
    await createFee(admin, offeringId, { amount: '153.00' });
    await admin.ctx.dispose();
    const { ctx, email, password } = await registerAndLoginStudent();
    await ctx.dispose();
    const login = await page.request.post('/api/v1/auth/login/', { data: { email, password } });
    const csrf = login.headers()['x-csrf-token'] || '';
    const { applicationId } = await createDraftApplication(page.request, csrf, {
      canonicalProgramId: canonical.id as string,
      intakeId,
      email,
    });
    const draftPath = `/api/v1/applications/drafts/${applicationId}/`;
    await page.context().addInitScript(() => {
      const log: string[] = [];
      (window as unknown as { __rl: string[] }).__rl = log;
      const t = () => Math.round(performance.now());
      document.addEventListener('visibilitychange', () => log.push(`${t()} visibility=${document.visibilityState}`));
      window.addEventListener('beforeunload', () => {
        log.push(`${t()} beforeunload (visibility=${document.visibilityState})`);
        sessionStorage.setItem('__rl_prev', JSON.stringify(log));
      });
    });
    await page.goto(`/student/application-wizard?mode=resume&draftId=${applicationId}`, { waitUntil: 'domcontentloaded' });
    await expect(page.getByTestId('wizard-action-bar')).toBeVisible({ timeout: 60_000 });
    await openPersonalStep(page);
    await page.waitForLoadState('networkidle');
    // A second tab of the same browser becomes the foreground tab, as in the two-tab journey.
    const other = await page.context().newPage();
    await other.goto('/', { waitUntil: 'domcontentloaded' });
    const visibility = await page.evaluate(() => document.visibilityState);
    const current = unwrap(await (await page.request.get(draftPath)).json());
    await page.request.patch(draftPath, {
      headers: { 'x-csrf-token': csrf, 'Idempotency-Key': `rb-${Date.now()}` },
      data: { residence_town: `Reload bg winner ${Date.now()}`, expected_version: Number(current.version) },
    });
    const stale = page.waitForResponse(
      (r) => r.request().method() === 'PATCH' && new URL(r.url()).pathname === draftPath,
      { timeout: 30_000 },
    );
    await page.getByLabel(/town|city/i).first().fill(`Reload bg stale ${Date.now()}`);
    await stale;
    const reloadLatest = page.getByRole('button', { name: /reload latest draft/i });
    await expect(reloadLatest).toBeEnabled();
    await page.evaluate(() => {
      const log = (window as unknown as { __rl: string[] }).__rl;
      document.addEventListener('click', () => log.push(`${Math.round(performance.now())} click`), { capture: true });
    });
    const startedAt = Date.now();
    const navigated = page.waitForNavigation({ waitUntil: 'domcontentloaded' });
    await reloadLatest.click();
    const clickReturned = Date.now() - startedAt;
    await navigated;
    const navigatedAfter = Date.now() - startedAt;
    const previous = await page.evaluate(() => sessionStorage.getItem('__rl_prev'));
    console.log(`DIAG-BG #${repeat}: tab visibility while the other tab is open = ${visibility}; click() returned after ${clickReturned} ms; navigation committed after ${navigatedAfter} ms`);
    console.log(`DIAG-BG #${repeat}: page log before unload: ${previous}`);
    await other.close();
  });
}

      const ric = window.requestIdleCallback?.bind(window);
      if (ric) {
        window.requestIdleCallback = ((cb: IdleRequestCallback, opts?: IdleRequestOptions) => {
          const at = t();
          return ric((deadline) => {
            if ((window as unknown as { __armed?: boolean }).__armed) log.push(`${t()} rIC fired (scheduled ${at})`);
            cb(deadline);
          }, opts);
        }) as typeof window.requestIdleCallback;
      }
      const reload = Location.prototype.reload;
      try {
        Object.defineProperty(window.location, 'reload', {
          configurable: true,
          value: () => {
            log.push(`${t()} location.reload() called`);
            sessionStorage.setItem('__rl_prev', JSON.stringify(log));
            reload.call(window.location);
          },
        });
      } catch {
        log.push('could not wrap location.reload');
      }
      const setItem = Storage.prototype.setItem;
      Storage.prototype.setItem = function (key: string, value: string) {
        if ((window as unknown as { __armed?: boolean }).__armed && key.includes('applicationWizardDraft:')) {
          log.push(`${t()} setItem ${key.replace(/:[0-9a-f-]{36}/g, ':<id>')}`);
        }
        return setItem.call(this, key, value);
      };
      const removeItem = Storage.prototype.removeItem;
      Storage.prototype.removeItem = function (key: string) {
        if ((window as unknown as { __armed?: boolean }).__armed && key.includes('applicationWizardDraft:')) {
          log.push(`${t()} removeItem ${key.replace(/:[0-9a-f-]{36}/g, ':<id>')}`);
        }
        return removeItem.call(this, key);
      };
      window.addEventListener('beforeunload', (event) => {
        setTimeout(() => undefined, 0);
        log.push(`${t()} beforeunload (defaultPrevented=${event.defaultPrevented})`);
        sessionStorage.setItem('__rl_prev', JSON.stringify(log));
      });
    });

    // Other device wins.
    await page.goto(`/student/application-wizard?mode=resume&draftId=${applicationId}`, { waitUntil: 'domcontentloaded' });
